country_event = {
    id = 69108
    title = "EVTNAME69108"
    desc = "EVTDESC69108"
    
    trigger = {
        tag = SPA
        NOT = { has_country_modifier = spain_vassal_tax_timer }
        OR = {
            is_our_vassal = NSP
            is_our_vassal = NGR
            is_our_vassal = VPR
            is_our_vassal = RDP
        }
    }
    immediate = {
        set_country_flag = viceroyalties_owned
        add_country_modifier = { name = spain_vassal_tax_timer duration = 360 }
        any_country = {
            limit = {
				OR = {
					tag = NGR
					tag = NSP
					tag = VPR
					tag = RDP
				}

            }
            random_owned = {
                limit = {
                    owner = {
                        vassal_of = SPA
                    }
                }
                owner = { 
                    country_event = { id = 69109 days = 0 }
                }
            }
        }
    }

    option = {
        name = "EVTOPTA69108"
        
    }
}
