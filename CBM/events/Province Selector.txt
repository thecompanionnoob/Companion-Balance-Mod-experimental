province_event = {
	id = 69029
	title = "EVTNAME69029"
	desc = "EVTDESC69029"
	
	is_triggered_only = yes
	
	allow_multiple_instances = no
	
	option = { #Attract immigration
		name = "EVTOPTA69029"
		owner = {
			any_owned = {
				limit = {
					has_building = province_selector
				}
				province_selector = -1
			}
		}
		add_core = JAN
		owner = {
			random_owned = {
				limit = {
					is_core = JAN
					NOT = { has_province_modifier = province_attract_immigrants }
				}
				owner = {
					set_country_flag = province_migration
					any_owned = {
						remove_province_modifier = province_attract_immigrants
					}
				}
				state_scope = { 
					any_owned = {
						add_province_modifier = { name = province_attract_immigrants duration = -1 }
					}
				}
			}
		}
		remove_core = JAN
	}
	
	option = { #Assimilate
		name = "EVTOPTB69029"
		owner = {
			any_owned = {
				limit = {
					has_building = province_selector
				}
				province_selector = -1
			}
		}
		add_core = JAN
		owner = {
			random_owned = {
				limit = {
					is_core = JAN
					NOT = { has_province_modifier = assim_focus }
					NOT = { is_colonial = yes }
				}
				owner = {
					set_country_flag = assim_focusing
					any_owned = {
						remove_province_modifier = assim_focus
						remove_core = ASS
						state_scope = { flashpoint_tension = -100 }
					}
				}
				state_scope = { 
					flashpoint_tension = 5
					any_owned = {
						add_province_modifier = { name = assim_focus duration = -1 }
						add_core = ASS
					}
				}
			}
		}
		remove_core = JAN
	}
	option = { #build research center
		name = "EVTOPTF69029"
		owner = {
			any_owned = {
				limit = {
					has_building = province_selector
				}
				province_selector = -1
			}
		}
		add_core = JAN
		owner = {
			random_owned = {
				limit = {
					is_core = JAN
					NOT = { 
						OR = {
							is_core = REA
							has_province_modifier = research_center
						}
					}
					owner = { check_variable = { which = research_center_points value = 0.99 } }
				}
				owner = { 
					change_variable = { which = research_center_points value = -1 }
				}
				add_province_modifier = {
					name = research_center
					duration = -1
				}
				add_core = REA

			}
		}
		remove_core = JAN 
	}
	option = { #supply depot 
		name = "EVTOPTE69029"
		owner = {
			any_owned = {
				limit = {
					has_building = province_selector
				}
				province_selector = -1
			}
		}
		add_core = JAN
		owner = {
			random_owned = {
				limit = {
					is_core = JAN
					NOT = { 
						has_province_modifier = prov_supply_depot
					}
					owner = { money = 500000 }
				}
				owner = { treasury = -500000 }
				owner = {
					any_owned = {
						remove_province_modifier = prov_supply_depot
					}
				}
				add_province_modifier = {
					name = prov_supply_depot
					duration = -1
				}

			}
		}
		remove_core = JAN 
	}
	option = { #nevermind
		name = "EVTOPTC69029"
		owner = {
			any_owned = {
				limit = {
					has_building = province_selector
				}
				province_selector = -1
			}
		}
	}
}

province_event = { #cleanup for supply depot
	id = 69098
	title = "EVTNAME69098"
	desc = "EVTDESC69098"

	trigger = {
		has_province_modifier = prov_supply_depot
		OR = {

		}

		remove_province_modifier = prov_supply_depot
	}

	option = {
		name = "EVTOPTA69098"
	}
